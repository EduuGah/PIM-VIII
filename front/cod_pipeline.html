jobs:
# ---------------------------------------------------------------- CI
  build-e-testes:
    name: Build, testes e analise estatica
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0 }

      - name: Configurar .NET 8
        uses: actions/setup-dotnet@v4
        with: { dotnet-version: '8.0.x' }

      - name: Restaurar dependencias
        run: dotnet restore AlunosTech.sln

      - name: Compilar em modo Release
        run: dotnet build --no-restore -c Release

      - name: Executar testes com cobertura
        run: |
          dotnet test --no-build -c Release \
            --collect:"XPlat Code Coverage" \
            --logger trx --results-directory ./resultados

      - name: Quality gate SonarQube (minimo 80% de cobertura)
        uses: sonarsource/sonarqube-quality-gate-action@v1
        env:
          SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}

      - name: Varredura de vulnerabilidades em dependencias
        run: dotnet list package --vulnerable --include-transitive
